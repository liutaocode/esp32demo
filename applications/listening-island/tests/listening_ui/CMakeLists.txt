cmake_minimum_required(VERSION 3.16)
project(listening_preview C)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${REPO_ROOT}/managed_components/lvgl__lvgl")
 get_filename_component(LVGL_DIR "${REPO_ROOT}/../../../managed_components/lvgl__lvgl" ABSOLUTE)
else()
 set(LVGL_DIR "${REPO_ROOT}/managed_components/lvgl__lvgl")
endif()
add_subdirectory("${LVGL_DIR}" lvgl)
add_executable(preview preview.c "${REPO_ROOT}/main/apps/listening/listening_state.c"
 "${REPO_ROOT}/main/apps/listening/listening_catalog.c" "${REPO_ROOT}/main/ui_pixel.c"
 "${REPO_ROOT}/assets/fonts/listening_zh_16.c" "${REPO_ROOT}/assets/fonts/listening_zh_22.c")
target_include_directories(preview PRIVATE "${REPO_ROOT}/main" "${REPO_ROOT}/main/apps/listening")
target_compile_options(preview PRIVATE -Wall -Wextra -Werror)
target_link_libraries(preview lvgl)
